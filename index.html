// --- Initialisation des variables globales --- let isRunning = false; let cameraUtils = null; let currentVelocity = 0; let lastExpression = "Neutre"; const videoElement = document.getElementById('input-video'); const canvasElement = document.getElementById('output-canvas'); const canvasCtx = canvasElement.getContext('2d'); const valFacsEl = document.getElementById('val-facs'); // --- Initialisation Three.js --- const container = document.getElementById('three-container'); const scene = new THREE.Scene(); const camera = new THREE.PerspectiveCamera(60, container.clientWidth / container.clientHeight, 0.1, 1000); const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true }); const material = new THREE.PointsMaterial({ size: 0.015, color: 0x10b981 }); const particlesMesh = new THREE.Points(geometry, material); scene.add(particlesMesh); function animate() { requestAnimationFrame(animate); const dynamicFactor = 1 + (currentVelocity * 200); particlesMesh.rotation.y += 0.001 * dynamicFactor; particlesMesh.rotation.x += 0.0005 * dynamicFactor; renderer.render(scene, camera); } animate(); // --- Logique MediaPipe & Analyse --- let expressionBuffer = []; const BUFFER_SIZE = 5; const faceMesh = new FaceMesh({ locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/face_mesh/${file}` }); faceMesh.setOptions({ maxNumFaces: 1, refineLandmarks: true, minDetectionConfidence: 0.5, minTrackingConfidence: 0.5 }); faceMesh.onResults((results) => { if (!isRunning) return; canvasCtx.clearRect(0, 0, canvasElement.width, canvasElement.height); canvasCtx.drawImage(results.image, 0, 0, canvasElement.width, canvasElement.height); expressionBuffer.push(mouthOpenness); if (expressionBuffer.length > BUFFER_SIZE) expressionBuffer.shift(); if (expressionBuffer.length === BUFFER_SIZE) { currentVelocity = Math.abs(expressionBuffer[BUFFER_SIZE - 1] - expressionBuffer[0]); // LOGIQUE DE DÉTECTION if (mouthOpenness > 0.05) { // MACRO-EXPRESSION valFacsEl.style.color = "#10b981"; valFacsEl.textContent = "Macro Active"; updateReport("Expression Ouverte", "Action intentionnelle détectée (parole ou rire)."); } else if (currentVelocity > 0.002) { // MICRO-EXPRESSION valFacsEl.style.color = "#fbbf24"; valFacsEl.textContent = (currentVelocity * 1000).toFixed(2); updateReport("Micro-Variation", "Réaction subconsciente ou stress émotionnel."); } else { valFacsEl.style.color = "#94a3b8"; valFacsEl.textContent = "0.00"; } } } }); // --- Fonction de rapport final en temps réel --- function updateReport(type, meaning) { if (lastExpression !== type) { const logEntry = `[ANALYSE] Type: ${type} | Signification: ${meaning}`; } } // --- Contrôles --- document.getElementById('btn-play').addEventListener('click', async () => { isRunning = true; document.getElementById('video-overlay').classList.add('hidden'); cameraUtils = new Camera(videoElement, { onFrame: async () => { if (isRunning) await faceMesh.send({ image: videoElement }); }, width: 640, height: 480 }); await cameraUtils.start(); }); document.getElementById('btn-stop').addEventListener('click', () => { isRunning = false; if (cameraUtils) { const stream = videoElement.srcObject;
