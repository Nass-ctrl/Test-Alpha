using UnityEngine; using System; using System.IO; using System.Collections.Generic; [Serializable] public class SessionReport { public string sessionID; public string timestamp; public string globalStatus; public CriticalThresholdAnalysis thresholds; public EmotionalState emotionalState; public TelemetryData telemetry; } [Serializable] public class CriticalThresholdAnalysis { public MetricStatus cognitiveLoad; public MetricStatus hrv; public MetricStatus blinkRate; public MetricStatus muscleTension; } [Serializable] public class MetricStatus { public string metricName; public float recordedValue; public float alertThreshold; public string status; // 🟢 Stable, 🟡 Attention, 🔴 Critique } [Serializable] public class EmotionalState { public string dominantEmotion; public string secondaryEmotion; public int microExpressionsDetected; public float valenceIndex; // -1.0 to 1.0 public float arousalIndex; // 0.0 to 1.0 } [Serializable] public class TelemetryData { public int avgHeartRateBPM; public int peakHeartRateBPM; public float estimatedSpO2; public int respiratoryRate; public float avgPupilDilation; public int fixationDurationMs; public float jawOpenBlendshape; public float mouthSmileBlendshape; public float browInnerUpBlendshape; } public class EgoLensSessionManager : MonoBehaviour { [Header("Session Info")] public string sessionID = "#8492-ALPHA"; private DateTime sessionStartTime; [Header("Live Metrics (Feed these from AR/ML modules)")] public float currentCognitiveLoad = 0f; public float currentHRV = 45f; public float currentBlinkRate = 20f; public float currentMuscleTension = 0f; public int currentBPM = 72; public float currentSpO2 = 98f; public float currentPupilDilation = 3.2f; // Trackers for Max/Min values during the session private float maxCognitiveLoad = 0f; private float minHRV = 100f; private float maxMuscleTension = 0f; private int peakBPM = 0; private int microExpressionCount = 0; void Start() { sessionStartTime = DateTime.Now; Debug.Log($"[EgoLens Apex-Spectra] Session {sessionID} démarrée."); } void Update() { // Enregistrement continu des extrêmes pour le rapport final if (currentCognitiveLoad > maxCognitiveLoad) maxCognitiveLoad = currentCognitiveLoad; if (currentHRV < minHRV) minHRV = currentHRV; if (currentMuscleTension > maxMuscleTension) maxMuscleTension = currentMuscleTension; if (currentBPM > peakBPM) peakBPM = currentBPM; } /// /// À appeler depuis votre module d'analyse faciale (FACS) à chaque détection. /// public void RegisterMicroExpression() { microExpressionCount++; } /// /// Génère et sauvegarde le rapport JSON. À appeler via un bouton UI ou OnApplicationQuit. /// public void EndSessionAndExportReport() { SessionReport finalReport = GenerateReport(); // Convertir l'objet en JSON formaté string json = JsonUtility.ToJson(finalReport, true); // Définir le chemin de sauvegarde (Fonctionne sur PC, iOS et Android) string fileName = $"EgoLens_Report_{sessionID}_{DateTime.Now:yyyyMMdd_HHmmss}.json"; string path = Path.Combine(Application.persistentDataPath, fileName); // Écrire le fichier File.WriteAllText(path, json); Debug.Log($"[EgoLens Apex-Spectra] Rapport sauvegardé avec succès : {path}"); } private SessionReport GenerateReport() { SessionReport report = new SessionReport { sessionID = sessionID, timestamp = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss"), globalStatus = DetermineGlobalStatus(), thresholds = new CriticalThresholdAnalysis { cognitiveLoad = new MetricStatus { metricName = "Charge Cognitive", recordedValue = maxCognitiveLoad, alertThreshold = 25f, status = GetStatus(maxCognitiveLoad, 25f, false) }, hrv = new MetricStatus { metricName = "Variabilité Cardiaque (HRV)", recordedValue = minHRV, alertThreshold = 30f, status = GetStatus(minHRV, 30f, true) }, blinkRate = new MetricStatus { metricName = "Fréquence Clignement", recordedValue = currentBlinkRate, alertThreshold = 35f, status = GetStatus(currentBlinkRate, 35f, false) }, muscleTension = new MetricStatus { metricName = "Tension Musculaire (AU4+AU7)", recordedValue = maxMuscleTension, alertThreshold = 50f, status = GetStatus(maxMuscleTension, 50f, false) } }, emotionalState = new EmotionalState { dominantEmotion = "Engagement / Focalisation", secondaryEmotion = "Calme / Neutralité", microExpressionsDetected = microExpressionCount, valenceIndex = 0.4f, arousalIndex = 0.6f }, telemetry = new TelemetryData { avgHeartRateBPM = currentBPM, // À remplacer par une vraie moyenne glissante peakHeartRateBPM = peakBPM, estimatedSpO2 = currentSpO2, respiratoryRate = 14, avgPupilDilation = currentPupilDilation, fixationDurationMs = 450, jawOpenBlendshape = 0.02f, mouthSmileBlendshape = 0.18f, browInnerUpBlendshape = 0.05f } }; return report; } private string GetStatus(float value, float threshold, bool isLowerBetter) { // Logique de seuil : si isLowerBetter est vrai, passer sous le seuil est mauvais (ex: HRV) bool isCritical = isLowerBetter ? value < threshold : value > threshold; bool isWarning = isLowerBetter ? value < threshold * 1.2f : value > threshold * 0.8f; if (isCritical) return "🔴 Critique"; if (isWarning) return "🟡 Attention"; return "🟢 Stable"; } private string DetermineGlobalStatus() { if (maxMuscleTension > 50f || minHRV < 30f) return "Tolérance nominale, avec pics d'activation ciblés."; return "Opérationnel et Stable."; } // Sauvegarde automatique si l'application est fermée brutalement private void OnApplicationQuit() { EndSessionAndExportReport(); } }
